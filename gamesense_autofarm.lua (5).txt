--[[
    Hallowen Candy Farm (standalone)
    Custom platform/teleport autofarm with a WindUI front end.
    Controls: farm, murderer avoidance, auto reset, bag-full fling,
    3D rendering, and an editable coin interval.
    All toggles start OFF. RightShift toggles the window.
]]

-- autofarm settings (same defaults as the previous build)
local Config = {
    AutoFarm = false, AutoReset = false, FarmAvoidMurderer = false,
    FlingAfterFarm = false, CoinRadius = 1000, CoinInterval = 2,
    CoinHoldDelay = 1, EngineMode = "TP Mode", GamesenseMovement = "Underground",
    GamesenseSpeed = 25, GamesenseCoinLimit = 40, GamesenseDownDepth = 14,
    GamesenseRiseXZ = 4, GamesenseAvoidDistance = 40, GamesenseSafeRiseDistance = 20,
}

-- ═══════════ CUSTOM TELEPORT AUTOFARM ═══════════
-- Creates a return platform above the active map, visits a coin, then returns
-- to the platform. The user-configurable interval controls the wait before
-- the next coin; bag-full detection and fling/reset remain separate.
local FlingGate = { busy = false }
local FarmAPI_TP = nil
local getMurderer

do
    local session = 0
    local running = false
    local phase, currentCoin = "idle", nil
    local lastCoin = nil
    local returnPlatform, platformMap, platformAnchor = nil, nil, nil
    local platformScanT, returnAt = 0, 0
    local lastReturnedAt = 0
    local coinContainers, containerScanT, containersScanned = {}, 0, false
    local bagFrame, bagFulls, bagScanT = nil, {}, 0
    local fullByServer = {}
    local skipT = {}
    local nextCoinAt = 0
    local lastRespawn = 0
    local farmErr = nil
    local murdererPosCache, murdererPosT = nil, 0
    local farmThread = nil
    local gamesenseTarget, gamesenseWasDown, gamesenseDownRefY = nil, false, nil
    local gamesenseNoclipCache, gamesenseSavedCharacter, gamesenseSavedHumanoid = {}, nil, nil
    local gamesenseSavedPlatformStand = nil
    local gamesenseLastTouch, gamesenseLastStep, gamesenseLastFallback = 0, 0, 0
    local gamesenseSawCoins, gamesenseFinished = false, false

    local function charModel()
        local ok, ch = pcall(function()
            return LocalPlayer.Character
        end)
        if not ok then return nil end
        return ch
    end

    local function rootPart()
        local ch = charModel()
        if not ch then return nil end
        local ok, rp = pcall(function()
            return ch:FindFirstChild("HumanoidRootPart")
        end)
        if not ok then return nil end
        return rp
    end

    local function humanoid()
        local ch = charModel()
        if not ch then return nil end
        local ok, hum = pcall(function()
            return ch:FindFirstChildOfClass("Humanoid")
        end)
        if not ok then return nil end
        return hum
    end

    local function isAlive()
        local hum = humanoid()
        if not hum then return false end
        local ok, hp = pcall(function()
            return hum.Health
        end)
        if not ok then return false end
        return (tonumber(hp) or 0) > 0
    end

    local function scanBag()
        local pg = LocalPlayer:FindFirstChild("PlayerGui")
        local main = pg and pg:FindFirstChild("MainGUI")
        local gameFrame = main and main:FindFirstChild("Game")
        bagFrame = gameFrame and gameFrame:FindFirstChild("CoinBags") or nil
        bagFulls = {}
        if not bagFrame then return end
        for _, obj in ipairs(bagFrame:GetDescendants()) do
            if obj.Name == "Full" and obj:IsA("GuiObject")
                and not obj:FindFirstAncestorWhichIsA("LuaSourceContainer") then
                bagFulls[#bagFulls + 1] = obj
            end
        end
    end

    local function bagFull()
        for _, isFull in pairs(fullByServer) do
            if isFull then return true end
        end
        if not bagFrame or not bagFrame.Parent or tick() - bagScanT > 2 then
            bagScanT = tick()
            scanBag()
        end
        if not bagFrame or not bagFrame.Visible then return false end
        for _, marker in ipairs(bagFulls) do
            if marker.Parent and marker.Visible then return true end
        end
        return false
    end

    local function listCoinContainers()
        if not containersScanned or tick() - containerScanT > 1 then
            containersScanned = true
            containerScanT = tick()
            coinContainers = {}
            for _, obj in ipairs(Workspace:GetDescendants()) do
                if obj.Name == "CoinContainer" then
                    coinContainers[#coinContainers + 1] = obj
                end
            end
        end
        return coinContainers
    end


    local function activeCoinMap()
        local bestContainer, bestMap, bestCount = nil, nil, -1
        for _, container in ipairs(listCoinContainers()) do
            if container.Parent then
                local count = 0
                for _, coin in ipairs(container:GetChildren()) do
                    if coin:IsA("BasePart") and coin:FindFirstChild("TouchInterest") then
                        count = count + 1
                    end
                end
                if count > bestCount then
                    bestContainer = container
                    bestMap = container.Parent
                    bestCount = count
                end
            end
        end
        if bestContainer and (not bestMap or bestMap == Workspace) then
            bestMap = bestContainer
        end
        return bestContainer, bestMap
    end

    local function computePlatformAnchor(container, map, fallback)
        local source = map
        if not source or source == Workspace then source = container end
        local minX, maxX, minY, maxY, minZ, maxZ
        local function includePart(part)
            if not part:IsA("BasePart") or part == returnPlatform then return end
            local pos = part.Position
            local size = part.Size or Vector3.new(0, 0, 0)
            local x1, x2 = pos.X - size.X / 2, pos.X + size.X / 2
            local y1, y2 = pos.Y - size.Y / 2, pos.Y + size.Y / 2
            local z1, z2 = pos.Z - size.Z / 2, pos.Z + size.Z / 2
            minX, maxX = math.min(minX or x1, x1), math.max(maxX or x2, x2)
            minY, maxY = math.min(minY or y1, y1), math.max(maxY or y2, y2)
            minZ, maxZ = math.min(minZ or z1, z1), math.max(maxZ or z2, z2)
        end
        if source then
            if source:IsA("BasePart") then includePart(source) end
            for _, obj in ipairs(source:GetDescendants()) do
                if obj:IsA("BasePart") then includePart(obj) end
            end
        end
        if not maxY then
            return Vector3.new(fallback.X, fallback.Y + 35, fallback.Z)
        end
        return Vector3.new((minX + maxX) / 2, maxY + 35, (minZ + maxZ) / 2)
    end

    local function ensureReturnPlatform(root)
        local now = tick()
        if not returnPlatform or not returnPlatform.Parent then
            returnPlatform = Workspace:FindFirstChild("AutoFarmReturnPlatform")
            if returnPlatform and not returnPlatform:IsA("BasePart") then
                returnPlatform:Destroy()
                returnPlatform = nil
            end
            if not returnPlatform then
                returnPlatform = Instance.new("Part")
                returnPlatform.Name = "AutoFarmReturnPlatform"
                returnPlatform.Size = Vector3.new(18, 1, 18)
                returnPlatform.Anchored = true
                returnPlatform.CanCollide = true
                pcall(function() returnPlatform.CanTouch = false end)
                returnPlatform.Transparency = 0.15
                returnPlatform.Color = Color3.fromRGB(80, 170, 255)
                returnPlatform.Parent = Workspace
            end
            platformMap, platformAnchor, platformScanT = nil, nil, 0
        end
        if now - platformScanT >= 1 or not platformAnchor then
            platformScanT = now
            local container, map = activeCoinMap()
            if map and map ~= platformMap then
                platformMap = map
                platformAnchor = computePlatformAnchor(container, map, root.Position)
            elseif not platformAnchor then
                platformMap = map
                platformAnchor = computePlatformAnchor(container, map, root.Position)
            end
        end
        if not platformAnchor then
            platformAnchor = Vector3.new(root.Position.X, root.Position.Y + 35, root.Position.Z)
        end
        if (returnPlatform.Position - platformAnchor).Magnitude > 0.1 then
            returnPlatform.CFrame = CFrame.new(platformAnchor)
            returnPlatform.Position = platformAnchor
        end
        return returnPlatform
    end
    local function platformStandPosition(platform)
        return platform.Position + Vector3.new(0, platform.Size.Y / 2 + 3, 0)
    end

    local function atPlatform(root, platform)
        return (root.Position - platformStandPosition(platform)).Magnitude <= 1
    end

    local function teleportToPlatform(root, platform)
        local position = platformStandPosition(platform)
        root.CFrame = CFrame.new(position)
        root.Position = position
        pcall(function()
            root.AssemblyLinearVelocity = Vector3.zero
            root.AssemblyAngularVelocity = Vector3.zero
        end)
    end

    local function coinInterval()
        return math.clamp(tonumber(Config.CoinInterval) or 2, 0.1, 60)
    end

    local function coinHoldDelay()
        return math.clamp(tonumber(Config.CoinHoldDelay) or 1, 0.05, 10)
    end

    local function murdererPos()
        if murdererPosT == 0 or os.clock() - murdererPosT > 0.25 then
            murdererPosT = os.clock()
            murdererPosCache = nil
            pcall(function()
                local mpl = getMurderer and getMurderer() or nil
                local rp = mpl and mpl.Character and mpl.Character:FindFirstChild("HumanoidRootPart")
                murdererPosCache = rp and rp.Position or nil
            end)
        end
        return murdererPosCache
    end

    local function coinUsable(coin, myPos)
        if not coin or not coin.Parent or not coin:IsA("BasePart") then return false end
        if not coin:FindFirstChild("TouchInterest") then return false end
        if (coin.Position - myPos).Magnitude > (tonumber(Config.CoinRadius) or 1000) then return false end
        return true
    end

    local function coinNearMurderer(coin, mPos)
        if not (coin and mPos) then return false end
        return Vector3.new(coin.Position.X - mPos.X, 0, coin.Position.Z - mPos.Z).Magnitude < 12
    end

    local function chooseCoin(myPos, mPos, excludeCoin)
        local best, bestDistance
        local now = tick()
        for _, container in ipairs(listCoinContainers()) do
            if container.Parent then
                for _, coin in ipairs(container:GetChildren()) do
                    if coin ~= excludeCoin and coinUsable(coin, myPos)
                        and (skipT[coin] or 0) <= now then
                        if coinNearMurderer(coin, mPos) then
                            skipT[coin] = now + 3
                        else
                            local distance = (coin.Position - myPos).Magnitude
                            if not bestDistance or distance < bestDistance then
                                best, bestDistance = coin, distance
                            end
                        end
                    end
                end
            end
        end
        return best
    end

    local function charTouchParts(ch)
        local parts = {}
        local head = ch:FindFirstChild("Head")
        if head then parts[#parts + 1] = head end
        for _, acc in ipairs(ch:GetChildren()) do
            if acc:IsA("Accessory") then
                local handle = acc:FindFirstChild("Handle")
                if handle then parts[#parts + 1] = handle end
            end
        end
        for _, name in ipairs({ "UpperTorso", "Torso", "HumanoidRootPart", "LowerTorso" }) do
            local part = ch:FindFirstChild(name)
            if part then parts[#parts + 1] = part end
        end
        return parts
    end

    local function touchCoin(ch, coin)
        if type(firetouchinterest) ~= "function" or not (ch and coin and coin.Parent) then return end
        for _, part in ipairs(charTouchParts(ch)) do
            pcall(function()
                firetouchinterest(part, coin, 0)
                firetouchinterest(part, coin, 1)
            end)
        end
    end

    local function teleportToCoin(root, coin)
        local position = coin.Position
        root.CFrame = CFrame.new(position)
        root.Position = position
        pcall(function()
            root.AssemblyLinearVelocity = Vector3.zero
            root.AssemblyAngularVelocity = Vector3.zero
        end)
        touchCoin(charModel(), coin)
    end

    local function autoRespawn()
        local hum = humanoid()
        if hum and hum.Health > 0 then
            hum.Health = 0
            lastRespawn = tick()
        end
    end

    local function gamesenseFlatDistance(a, b)
        return Vector3.new(a.X - b.X, 0, a.Z - b.Z).Magnitude
    end

    local function gamesenseCoinAvailable(coin, position)
        if not coin or not coin.Parent or not coin:IsA("BasePart") then return false end
        local okCollected, collected = pcall(function() return coin:GetAttribute("Collected") end)
        local okDelete, deleted = pcall(function() return coin:GetAttribute("Delete") end)
        if (okCollected and collected) or (okDelete and deleted) then return false end
        if position and (coin.Position - position).Magnitude > (tonumber(Config.CoinRadius) or 1000) then return false end
        return true
    end

    local function gamesenseCoins(position)
        local list, seen = {}, {}
        local function add(coin)
            if not seen[coin] and gamesenseCoinAvailable(coin, position) then
                seen[coin] = true
                list[#list + 1] = coin
            end
        end
        local ok, tagged = pcall(function() return CollectionService:GetTagged("CoinVisual") end)
        if ok and type(tagged) == "table" then
            for _, coin in ipairs(tagged) do add(coin) end
        end
        if #list == 0 then
            for _, container in ipairs(listCoinContainers()) do
                if container and container.Parent then
                    for _, coin in ipairs(container:GetChildren()) do add(coin) end
                end
            end
        end
        return list
    end

    local function gamesenseNearestCoin(position, list)
        local best, bestDistance = nil, math.huge
        for _, coin in ipairs(list) do
            local distance = (coin.Position - position).Magnitude
            if distance < bestDistance then best, bestDistance = coin, distance end
        end
        return best
    end

    local function gamesensePickCoin(position, list, murdererPosition)
        if not (Config.FarmAvoidMurderer and murdererPosition) then
            return gamesenseNearestCoin(position, list)
        end
        local avoidDistance = math.clamp(tonumber(Config.GamesenseAvoidDistance) or 40, 10, 100)
        local safe, safeDistance, farthest, farthestDistance = nil, math.huge, nil, -1
        for _, coin in ipairs(list) do
            local murdererDistance = gamesenseFlatDistance(coin.Position, murdererPosition)
            if murdererDistance > farthestDistance then
                farthest, farthestDistance = coin, murdererDistance
            end
            if murdererDistance >= avoidDistance then
                local distance = (coin.Position - position).Magnitude
                if distance < safeDistance then safe, safeDistance = coin, distance end
            end
        end
        if safe then return safe end
        if farthest and farthestDistance >= avoidDistance * 0.6 then return farthest end
        return nil
    end

    local function gamesenseAvoidSteer(current, destination, murdererPosition)
        if not (Config.FarmAvoidMurderer and murdererPosition) then return destination end
        local avoidDistance = math.clamp(tonumber(Config.GamesenseAvoidDistance) or 40, 10, 100)
        local murdererDistance = gamesenseFlatDistance(current, murdererPosition)
        if murdererDistance >= avoidDistance then return destination end
        local away = Vector3.new(current.X - murdererPosition.X, 0, current.Z - murdererPosition.Z)
        if away.Magnitude < 0.1 then away = Vector3.new(1, 0, 0) end
        away = away.Unit
        local desired = Vector3.new(destination.X - current.X, 0, destination.Z - current.Z)
        local distance = desired.Magnitude
        if distance < 0.1 then return destination end
        local weight = 1 + (1 - murdererDistance / avoidDistance) * 2
        local blend = desired.Unit + away * weight
        if blend.Magnitude < 0.1 then blend = away else blend = blend.Unit end
        local nextPosition = current + blend * distance
        return Vector3.new(nextPosition.X, destination.Y, nextPosition.Z)
    end

    local function gamesenseCoinTouches(coin)
        if type(firetouchinterest) ~= "function" or not coin or not coin.Parent then return end
        local now = os.clock()
        if now - gamesenseLastTouch < 0.05 then return end
        gamesenseLastTouch = now
        local root = rootPart()
        if not root then return end
        local targets, seen = {}, {}
        local function add(part)
            if not part or seen[part] then return end
            local ok, valid = pcall(function()
                return part:IsA("BasePart") and (part:FindFirstChildOfClass("TouchTransmitter") ~= nil
                    or part:FindFirstChild("TouchInterest") ~= nil)
            end)
            if ok and valid then seen[part] = true targets[#targets + 1] = part end
        end
        add(coin)
        for _, child in ipairs(coin:GetChildren()) do add(child) end
        local parent = coin.Parent
        if parent then
            add(parent)
            for _, child in ipairs(parent:GetChildren()) do add(child) end
        end
        if #targets == 0 then targets[1] = coin end
        for _, target in ipairs(targets) do
            pcall(function()
                firetouchinterest(root, target, 0)
                firetouchinterest(root, target, 1)
            end)
        end
    end

    local function gamesenseCurrentCoinCount()
        local ok, count = pcall(function()
            local playerGui = LocalPlayer:FindFirstChild("PlayerGui")
            local main = playerGui and playerGui:FindFirstChild("MainGUI")
            local gameGui = main and (main:FindFirstChild("Game") or main)
            local bags = gameGui and gameGui:FindFirstChild("CoinBags")
            local container = bags and bags:FindFirstChild("Container")
            local coin = container and container:FindFirstChild("Coin")
            local currency = coin and coin:FindFirstChild("CurrencyFrame")
            local icon = currency and currency:FindFirstChild("Icon")
            local label = icon and icon:FindFirstChild("Coins")
            if not label then return 0 end
            local text = tostring(label.Text or "")
            return tonumber(text) or tonumber(string.match(text, "%d+")) or 0
        end)
        return ok and count or 0
    end

    local function gamesenseReturnToSurface(root)
        if not gamesenseWasDown then return end
        root = root or (gamesenseSavedCharacter and gamesenseSavedCharacter:FindFirstChild("HumanoidRootPart")) or rootPart()
        if not root then gamesenseWasDown = false return end
        local origin = root.Position
        local y = nil
        pcall(function()
            local params = RaycastParams.new()
            params.FilterType = Enum.RaycastFilterType.Exclude
            params.FilterDescendantsInstances = { gamesenseSavedCharacter or charModel() }
            params.IgnoreWater = true
            local result = Workspace:Raycast(origin, Vector3.new(0, 400, 0), params)
            if result then y = result.Position.Y + 5 end
        end)
        if not y and gamesenseDownRefY then y = gamesenseDownRefY + 5 end
        if y then
            local position = Vector3.new(origin.X, y, origin.Z)
            pcall(function()
                root.CFrame = CFrame.new(position)
                root.Position = position
                root.AssemblyLinearVelocity = Vector3.zero
                root.AssemblyAngularVelocity = Vector3.zero
            end)
        end
        gamesenseWasDown = false
    end

    local function gamesenseRestore(root)
        gamesenseReturnToSurface(root)
        local humanoidToRestore = gamesenseSavedHumanoid or humanoid()
        if humanoidToRestore and gamesenseSavedPlatformStand ~= nil then
            pcall(function() humanoidToRestore.PlatformStand = gamesenseSavedPlatformStand end)
        end
        for part, canCollide in pairs(gamesenseNoclipCache) do
            if part and part.Parent then pcall(function() part.CanCollide = canCollide end) end
        end
        gamesenseNoclipCache = {}
        gamesenseSavedCharacter, gamesenseSavedHumanoid, gamesenseSavedPlatformStand = nil, nil, nil
    end

    local function gamesenseSetNoclip(on)
        if not on then gamesenseRestore()
        else
            local character = charModel()
            local hum = humanoid()
            if not character then return end
            if gamesenseSavedCharacter and gamesenseSavedCharacter ~= character then
                gamesenseRestore(gamesenseSavedCharacter:FindFirstChild("HumanoidRootPart"))
            end
            if not gamesenseSavedCharacter then
                gamesenseSavedCharacter = character
                gamesenseSavedHumanoid = hum
                gamesenseSavedPlatformStand = hum and hum.PlatformStand or false
            end
            if hum then pcall(function() hum.PlatformStand = true end) end
            for _, part in ipairs(character:GetDescendants()) do
                if part:IsA("BasePart") then
                    if gamesenseNoclipCache[part] == nil then gamesenseNoclipCache[part] = part.CanCollide end
                    part.CanCollide = false
                end
            end
        end
    end

    local function gamesenseMove(root, destination, dt)
        local direction = destination - root.Position
        local distance = direction.Magnitude
        local nextPosition = destination
        if distance > 0.1 then
            local speed = math.clamp(tonumber(Config.GamesenseSpeed) or 25, 10, 50)
            nextPosition = root.Position + direction.Unit * math.min(speed * dt, distance)
        end
        local frame = CFrame.new(nextPosition)
        if Config.GamesenseMovement == "Underground" then
            frame = frame * CFrame.Angles(-math.pi * 0.5, 0, 0)
            gamesenseWasDown = true
        end
        pcall(function()
            root.CFrame = frame
            root.Position = nextPosition
            root.AssemblyLinearVelocity = Vector3.zero
            root.AssemblyAngularVelocity = Vector3.zero
        end)
    end

    local function gamesenseStep(root, now)
        local elapsed = gamesenseLastStep > 0 and now - gamesenseLastStep or 0.05
        local dt = math.clamp(elapsed, 0.01, 0.25)
        gamesenseLastStep = now

        if bagFull() then
            gamesenseTarget = nil
            gamesenseRestore(root)
            phase = "full"
            return
        end
        if FlingGate.busy then
            gamesenseTarget = nil
            gamesenseRestore(root)
            phase = "busy"
            return
        end

        local coinCount = gamesenseCurrentCoinCount()
        local coinLimit = math.clamp(tonumber(Config.GamesenseCoinLimit) or 40, 10, 50)
        if Config.AutoReset and coinCount >= coinLimit and now - lastRespawn > 3 then
            gamesenseTarget = nil
            gamesenseRestore(root)
            phase = "full"
            autoRespawn()
            return
        end

        local list = gamesenseCoins(root.Position)
        if #list == 0 then
            gamesenseTarget = nil
            gamesenseRestore(root)
            phase = "wait"
            if gamesenseSawCoins and not gamesenseFinished and Config.AutoReset and coinCount > 0
                and now - lastRespawn > 3 then
                gamesenseFinished = true
                autoRespawn()
            end
            return
        end
        gamesenseSawCoins = true
        gamesenseFinished = false

        local murdererPosition = nil
        if Config.FarmAvoidMurderer then
            local murdererRoot = murdererPos()
            murdererPosition = murdererRoot and murdererRoot.Position or nil
        end
        if not gamesenseCoinAvailable(gamesenseTarget, root.Position)
            or (gamesenseTarget and Config.FarmAvoidMurderer and murdererPosition
                and gamesenseFlatDistance(gamesenseTarget.Position, murdererPosition)
                    < math.clamp(tonumber(Config.GamesenseAvoidDistance) or 40, 10, 100) * 0.6) then
            gamesenseTarget = gamesensePickCoin(root.Position, list, murdererPosition)
        end

        if gamesenseTarget then
            gamesenseSetNoclip(true)
            local coinPosition = gamesenseTarget.Position
            gamesenseDownRefY = coinPosition.Y
            local destination = coinPosition
            if Config.GamesenseMovement == "Underground" then
                local xzDistance = gamesenseFlatDistance(root.Position, coinPosition)
                local safeRiseDistance = math.clamp(tonumber(Config.GamesenseSafeRiseDistance) or 20, 5, 60)
                local safeToRise = not murdererPosition
                    or gamesenseFlatDistance(root.Position, murdererPosition) > safeRiseDistance
                if xzDistance <= math.clamp(tonumber(Config.GamesenseRiseXZ) or 4, 1, 10) and safeToRise then
                    destination = coinPosition
                    gamesenseCoinTouches(gamesenseTarget)
                else
                    local depth = math.clamp(tonumber(Config.GamesenseDownDepth) or 14, 5, 40)
                    destination = Vector3.new(coinPosition.X, coinPosition.Y - depth, coinPosition.Z)
                end
            else
                if gamesenseWasDown then gamesenseReturnToSurface(root) end
                if (coinPosition - root.Position).Magnitude <= 6 then
                    gamesenseCoinTouches(gamesenseTarget)
                end
            end
            destination = gamesenseAvoidSteer(root.Position, destination, murdererPosition)
            gamesenseMove(root, destination, dt)
            phase = "gamesense-moving"
            return
        end

        if murdererPosition then
            gamesenseSetNoclip(true)
            local away = Vector3.new(root.Position.X - murdererPosition.X, 0, root.Position.Z - murdererPosition.Z)
            if away.Magnitude < 0.1 then away = Vector3.new(1, 0, 0) end
            away = away.Unit
            local y = root.Position.Y
            if Config.GamesenseMovement == "Underground" and gamesenseDownRefY then
                y = gamesenseDownRefY - math.clamp(tonumber(Config.GamesenseDownDepth) or 14, 5, 40)
            end
            gamesenseMove(root, root.Position + away * 40 + Vector3.new(0, y - root.Position.Y, 0), dt)
            phase = "gamesense-avoiding"
        else
            gamesenseRestore(root)
            phase = "wait"
        end
    end

    local function setEngineMode(mode)
        assert(mode == "TP Mode" or mode == "Gamesense Mode", "Unknown farm mode")
        if Config.EngineMode == mode then return false end
        if Config.EngineMode == "Gamesense Mode" then gamesenseRestore(rootPart()) end
        if returnPlatform then pcall(function() returnPlatform:Destroy() end) end
        returnPlatform, platformMap, platformAnchor = nil, nil, nil
        gamesenseTarget = nil
        currentCoin, lastCoin = nil, nil
        returnAt, lastReturnedAt, nextCoinAt = 0, 0, 0

function api:SetTuning(key, value)
    local spec = tuningSpecs[key]
    assert(spec, "Unknown tuning setting: " .. tostring(key))
    value = tonumber(value)
    if not value then
        syncControl(self.Controls[spec.control], tostring(self.State[spec.state]))
        return false
    end
    value = math.clamp(value, spec.min, spec.max)
    value = math.floor(value / spec.step + 0.5) * spec.step
    if spec.precision == 2 then value = math.floor(value * 100 + 0.5) / 100
    elseif spec.precision == 1 then value = math.floor(value * 10 + 0.5) / 10
    else value = math.floor(value + 0.5) end
    local changedValue = self.State[spec.state] ~= value
    self.State[spec.state] = value
    Config[spec.config] = value
    syncControl(self.Controls[spec.control], value)
    if changedValue then changed:Fire(key, value) end
    return true
end

function api:SelectTab(index)
    assert(type(index) == "number" and index % 1 == 0 and index >= 0 and index <= 1,
        "Tab must be 0..1")
    self.Tab = index
    local tab = self.Tabs[index + 1]
    if tab and type(tab.Select) == "function" then tab:Select() end
end

local function addToggle(section, key, title, description, iconName)
    local control = section:Toggle({
        Title = title,
        Desc = description,
        Icon = iconName,
        Value = false,
        Callback = function(value)
            if not syncingControls then api:Set(key, value == true) end
        end,
    })
    api.Controls[key] = control
    return control
end

local farm_stop_all

    if FarmAPI_TP then pcall(FarmAPI_TP.stop) end
end

    if FarmAPI_TP then pcall(FarmAPI_TP.start) end
end
api.Changed:Connect(function(key, value)
    if key == "autoFarm" then
        Config.AutoFarm = value
        if value then
            task.spawn(function() farm_start_current() end)
        else
            farm_stop_all()
        end
    elseif key == "murderCheck" then
        Config.FarmAvoidMurderer = value
    elseif key == "autoReset" then
        Config.AutoReset = value
    elseif key == "renderOff" then
        pcall(function() RunService:Set3dRenderingEnabled(not value) end)
    elseif key == "flingMurder" then
        Config.FlingAfterFarm = value
    elseif key == "interval" then
        Config.CoinInterval = tonumber(value) or 2
    elseif key == "farmMode" then
        Config.EngineMode = value
    elseif key == "gamesenseMovement" then
        Config.GamesenseMovement = value
    elseif key == "coinHold" then
        Config.CoinHoldDelay = tonumber(value) or 1
    elseif key == "gamesenseSpeed" then
        Config.GamesenseSpeed = tonumber(value) or 25
    elseif key == "gamesenseCoinLimit" then
        Config.GamesenseCoinLimit = tonumber(value) or 40
    elseif key == "gamesenseDownDepth" then
        Config.GamesenseDownDepth = tonumber(value) or 14
    elseif key == "gamesenseRiseXZ" then
        Config.GamesenseRiseXZ = tonumber(value) or 4
    elseif key == "gamesenseAvoidDistance" then
        Config.GamesenseAvoidDistance = tonumber(value) or 40
    elseif key == "gamesenseSafeRiseDistance" then
        Config.GamesenseSafeRiseDistance = tonumber(value) or 20
    end
end)

-- auto fling murder: when the coin bags are full, fling the murderer until dead
do
    local flingAcc = 0
    local flingBusy = false
    local currentMurderer = nil
    RunService.Heartbeat:Connect(function(dt)
        flingAcc = flingAcc + (tonumber(dt) or 0)
        if flingAcc < 0.5 then return end
        flingAcc = 0
        local m = nil
        pcall(function() m = getMurderer() end)
        currentMurderer = m
        if not (Config.AutoFarm and Config.FlingAfterFarm) then return end
        local full = false
        pcall(function()
            full = FarmAPI_TP ~= nil and FarmAPI_TP.running() == true
                and FarmAPI_TP.bagFull ~= nil and FarmAPI_TP.bagFull() == true
        end)
        if not full then return end
        if flingBusy then return end
        if m then
            flingBusy = true
            FlingGate.busy = true
            task.spawn(function()
                while Config.AutoFarm and Config.FlingAfterFarm do
                    local stillAlive = false
                    pcall(function() stillAlive = (getMurderer() == m) end)
                    if not stillAlive then break end
                    pcall(do_fling, m, true)
                    task.wait(0.1)
                end
                FlingGate.busy = false
                flingBusy = false
            end)
        end
    end)
    if getgenv().FarmAPI then
        getgenv().FarmAPI.current_murderer = function() return currentMurderer end
    end
end

-- ALWAYS-ON watchdog
do
    local wdAcc = 0
    RunService.Heartbeat:Connect(function(dt)
        wdAcc = wdAcc + (tonumber(dt) or 0)
        if wdAcc < 1 then return end
        wdAcc = 0
        if not Config.AutoFarm then return end
        local alive = false
        pcall(function()
            alive = (FarmAPI_TP ~= nil and FarmAPI_TP.running() == true
                and FarmAPI_TP.thread() ~= nil
                and coroutine.status(FarmAPI_TP.thread()) ~= "dead")
        end)
        if not alive then
            pcall(farm_stop_all)
            task.spawn(function() pcall(farm_start_current) end)
        end
    end)
end
